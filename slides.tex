      \begin{block}{Decoder Block}
        \begin{itemize}
          \item RMSNorm
          \item Causal scaled dot-product softmax
          \item RoPE position rotation
          \item Grouped-query attention
          \item SwiGLU feed-forward
          \item Weight-tied output projection
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \bigskip
  \centering
  The neural model decides \alert{what kind of reply happens next};
  the website decides \alert{what words hit the chat box}.
\end{frame}

\begin{frame}{Softmax Attention}
  \[
    A = \operatorname{softmax}\!\left(\frac{QK^\top}{\sqrt{d_k}} + M\right),
    \qquad O = AV
  \]

  \begin{itemize}
    \item This is the only active attention path now.
    \item For this tiny model, softmax is faster/saner in practice.
    \item The checkpoint with the best measured validation accuracy is softmax.
    \item GQA keeps the browser checkpoint compact.
  \end{itemize}

  \bigskip
  \centering
  Boring math, useful model. Ship it.
\end{frame}

\begin{frame}{Parameter Count}
  \small
  Config: $V=15$, $N=1$, $d=16$, $n_h=4$, $n_{kv}=1$, $d_k=4$,
  $d_{ff}=16$

  \medskip
  \centering
  \begin{tabular}{llr}
    \toprule
    \textbf{Component} & \textbf{Formula} & \textbf{Params} \\
    \midrule
    Embedding & $V \times d = 15 \times 16$ & 240 \\
    Attention per layer & $d^2 + 2dn_{kv}d_k + d^2$ & 640 \\
    FFN per layer (SwiGLU) & $3dd_{ff}$ & 768 \\
    Norms per layer & $2d$ & 32 \\
    Final norm & $d$ & 16 \\
    \midrule
    Total & $240 + 1(640 + 768 + 32) + 16$ & \alert{1,696} \\
    \bottomrule
  \end{tabular}

  \medskip
  Output projection reuses the embedding matrix, so it adds zero new params.
\end{frame}

\begin{frame}{Training Path}
  \begin{block}{Data Build}
    \code{python examples/build\_action\_data.py --offline}
  \end{block}

  \begin{block}{CUDA Training}
    \small
    \code{cargo run --release --features cuda --example train\_action\_model --}\\
    \code{\quad --steps 20000 --batch-size 64 --lr 0.003}\\
    \code{\quad --checkpoint-dir runs/action-v4-d16-clean}
  \end{block}

  \begin{itemize}
    \item Training is CUDA-only by design.
    \item Website runtime can choose WebGPU or CPU.
    \item Training data is compact action-ID sequences.
  \end{itemize}

\section{Website}

\begin{frame}{Browser Runtime}
  \begin{itemize}
    \item Website loads \code{model-personal-v2.mpk}
    \item Website loads \code{model-config-personal-v2.json}
    \item Runtime tries WebGPU first when requested
    \item Auto mode falls back to CPU when WebGPU is unavailable
    \item Chrome without a valid WebGPU adapter should not kill the chat box
    \item UI streams message text from the response graph
  \end{itemize}

  \bigskip
  \begin{block}{Mood Contract}
    Model output action sequence maps to one mood:
    \code{normal}, \code{happy}, \code{frustrated}, or \code{sad}.
    The mascot GIF changes from that mood.
  \end{block}
\end{frame}

\begin{frame}{Deploy Shape}
  \begin{itemize}
    \item Build the Leptos site with \code{trunk build --release}
    \item Ship the generated \code{dist/} directory to Vercel
    \item Keep model assets in \code{dancing-with-my-code-v2/assets/}
    \item Cache \code{.mpk} and \code{.json} assets aggressively
  \end{itemize}

  \bigskip
  \begin{block}{Final State}
    Softmax drives the tiny action model. The website owns the voice.
  \end{block}
\end{frame}
