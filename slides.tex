\section{Normalization}
% ================================================================

% ---------------------------------------------------------------
\begin{frame}{RMSNorm}
  \begin{block}{Formula}
    \[
      \text{\color{normorg}RMSNorm}(\mathbf{x})
      \;=\;
      \frac{\mathbf{x}}{\sqrt{\operatorname{mean}(\mathbf{x}^{2}) + \varepsilon}}
      \;\odot\;
      \mathbf{w}
    \]
  \end{block}

  \medskip

  \begin{itemize}
    \item \textbf{No centering} (no mean subtraction) --- just root-mean-square scaling
    \item Learnable weight vector $\mathbf{w}$, initialized to $\mathbf{1.0}$
    \item $\varepsilon = 10^{-5}$ for numerical stability
  \end{itemize}

  \medskip

  \begin{exampleblock}{Code Reference}
    \ttfamily\small
    denom = x.square().mean\_dim(2).add\_scalar(eps).sqrt();\\
    output = (x / denom) * weight
  \end{exampleblock}
\end{frame}

\section{Positional Encoding}
% ================================================================

% ---------------------------------------------------------------
\begin{frame}{RoPE --- Concept}
  \textbf{Rotary Position Embedding} encodes position by
  \emph{rotating pairs of dimensions} in the embedding space.

  \bigskip

  \begin{itemize}
    \item No learned positional embeddings needed
    \item Base frequency $\theta = 10\,000$
    \item Per-pair frequency:
      \[
        \theta_k = \theta^{\,-2k / d_{\text{head}}},
        \qquad k = 0, 1, \ldots, \tfrac{d_{\text{head}}}{2} - 1
      \]
    \item Rotation applied to \textbf{Q} and \textbf{K} \emph{before} attention
    \item Relative position emerges naturally from dot products:
      $\mathbf{q}_m^{\!\top} \mathbf{k}_n$
      depends only on $m - n$
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{RoPE --- Formula}
  For each pair $(x_{2k},\; x_{2k+1})$ at position $m$:

  \begin{block}{2D Rotation by Angle $m\theta_k$}
    \[
      \begin{aligned}
        \text{RoPE}(\mathbf{x}, m)_{2k}
          &= x_{2k} \cos(m\theta_k)
           - x_{2k+1} \sin(m\theta_k) \\[6pt]
        \text{RoPE}(\mathbf{x}, m)_{2k+1}
          &= x_{2k} \sin(m\theta_k)
           + x_{2k+1} \cos(m\theta_k)
      \end{aligned}
    \]
  \end{block}

  \medskip
  \begin{itemize}
    \item Low-frequency dims (small $k$) rotate \textbf{slowly}
    \item High-frequency dims (large $k$) rotate \textbf{quickly}
    \item Equivalent to complex-element-wise multiplication:
      $x_k \cdot e^{i m\theta_k}$
  \end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{RoPE --- Worked Example}
  \small
  Config: $d_{\text{head}} = 16 \Rightarrow 8$ pairs \qquad Position $m = 3$

  \medskip

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \textbf{Frequencies}

      \medskip
      \begin{tabular}{cccl}
        \toprule
        $k$ & $\theta_k$ & $m\theta_k$ & \\
        \midrule
        0 & $10000^{0}      = 1.000$  & $3.000$ \\
        1 & $10000^{-0.125} \approx 0.316$ & $0.949$ \\
        2 & $10000^{-0.250} \approx 0.100$ & $0.300$ \\
        3 & $10000^{-0.375} \approx 0.032$ & $0.095$ \\
        \bottomrule
      \end{tabular}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \textbf{Rotation of pair $k{=}0$}

      \medskip
      Input pair: $(x_0, x_1) = (1.0,\; 0.5)$

      \[
        \begin{aligned}
          x'_0 &= 1.0 \cos 3.0 - 0.5 \sin 3.0 \\
               &\approx -1.12 \\
          x'_1 &= 1.0 \sin 3.0 + 0.5 \cos 3.0 \\
               &\approx 0.03
        \end{aligned}
      \]

      \medskip
      \small
      (Values rounded; $\cos 3 \approx -0.990$, $\sin 3 \approx 0.141$)
    \end{column}
  \end{columns}
\end{frame}

\section{Attention}
% ================================================================

% ---------------------------------------------------------------
\begin{frame}{GQA --- Concept}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Standard Multi-Head Attention (MHA):}
      \begin{itemize}
        \item Each of $n_{\text{heads}}$ heads has separate Q, K, V
      \end{itemize}

      \medskip
      \textbf{{\color{attnblue}Grouped-Query Attention (GQA):}}
      \begin{itemize}
        \item $n_{\text{kv}} < n_{\text{heads}}$
        \item K and V heads are \alert{shared} across groups
        \item KV heads repeated $\lceil n_{\text{heads}} / n_{\text{kv}} \rceil$
              times to match Q
      \end{itemize}

      \bigskip
      \textbf{Benefit:}
      \begin{itemize}
        \item Smaller KV cache
        \item Faster inference
        \item Fewer parameters
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[
        qbox/.style={draw, fill=attnblue!20, minimum size=0.5cm,
                     font=\scriptsize},
        kbox/.style={draw, fill=ffngreen!30, minimum size=0.5cm,
                     font=\scriptsize},
        >=Stealth, scale=0.9, transform shape
      ]
        % MHA
        \node[font=\small\bfseries] at (0, 3.3) {MHA};
        \foreach \i in {0,...,3} {
          \node[qbox] (q\i) at (\i*0.7, 2.7) {Q};
          \node[kbox] (k\i) at (\i*0.7, 2.0) {K\,\i};
          \draw[->, thin] (q\i) -- (k\i);
        }

        % GQA
        \node[font=\small\bfseries] at (0, 1.2) {GQA};
        \foreach \i in {0,...,3} {
          \node[qbox] (gq\i) at (\i*0.7, 0.6) {Q};
        }
        \node[kbox, minimum width=2.4cm] (gk) at (1.05, -0.15) {K\,(shared)};
        \foreach \i in {0,...,3} {
          \draw[->, thin] (gq\i) -- (gk);
        }
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{GQA --- Formula}
  \begin{block}{Scaled Dot-Product Attention with Causal Mask}
    \[
      {\color{attnblue}
      \operatorname{Attention}(\mathbf{Q}, \mathbf{K}, \mathbf{V})
      \;=\;
      \operatorname{softmax}\!\Bigl(
        \frac{\mathbf{Q}\mathbf{K}^{\!\top}}{\sqrt{d_k}}
        + \mathbf{M}
      \Bigr)
      \,\mathbf{V}}
    \]
  \end{block}

  \smallskip
  \begin{itemize}
    \item $\mathbf{M}$: causal mask ($-\infty$ for future positions)
    \item $\mathbf{Q}$: $(B,\; n_{\text{heads}},\; L,\; d_k)$
    \item $\mathbf{K}, \mathbf{V}$: $(B,\; n_{\text{kv}},\; L,\; d_k)$
          $\;\to\;$ repeated to match $n_{\text{heads}}$
  \end{itemize}

  \medskip
  \begin{block}{Weight Matrices}
    \[
      \mathbf{W}_Q \in \mathbb{R}^{d \times d},\;\;
      \mathbf{W}_K \in \mathbb{R}^{d \times n_{\text{kv}} \cdot d_k},\;\;
      \mathbf{W}_V \in \mathbb{R}^{d \times n_{\text{kv}} \cdot d_k},\;\;
      \mathbf{W}_O \in \mathbb{R}^{d \times d}
    \]
  \end{block}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{GQA --- Worked Example (Action Preset)}
  Config: $d{=}64$,\; $n_{\text{heads}}{=}4$,\;
  $n_{\text{kv}}{=}1$,\; $d_k{=}16$

  \bigskip
  \centering
  \begin{tabular}{llrl}
    \toprule
    \textbf{Matrix} & \textbf{Shape} & \textbf{Params} & \\
    \midrule
    $\mathbf{W}_Q$ & $64 \times 64$          & 4\,096 & \\
    $\mathbf{W}_K$ & $64 \times 16$          & 1\,024 & {\small only 1 KV head!} \\
    $\mathbf{W}_V$ & $64 \times 16$          & 1\,024 & \\
    $\mathbf{W}_O$ & $64 \times 64$          & 4\,096 & \\
    \midrule
    \textbf{Total per layer} & & \textbf{10\,240} & \\
    \bottomrule
  \end{tabular}

  \bigskip
  \alert{KV saving:}
  MHA would need $64{\times}64 + 64{\times}64 = 8\,192$ for K$+$V.\\
  GQA needs only $64{\times}16 + 64{\times}16 = 2\,048$
  --- \alert{75\,\% less!}
\end{frame}

\section{Feed-Forward Network}
% ================================================================

% ---------------------------------------------------------------
\begin{frame}{SwiGLU Feed-Forward Network}
  \begin{block}{Formula}
    \[
      {\color{ffngreen}
      \operatorname{FFN}(\mathbf{x})
      \;=\;
      \Bigl(
        \operatorname{SiLU}(\mathbf{x}\,\mathbf{W}_{\text{gate}})
        \;\odot\;
        \mathbf{x}\,\mathbf{W}_{\text{up}}
      \Bigr)
      \;\mathbf{W}_{\text{down}}}
    \]
  \end{block}

  \smallskip
  where \; $\operatorname{SiLU}(z) = z \cdot \sigma(z)$, \;
  $\sigma(z) = \frac{1}{1 + e^{-z}}$ \; (sigmoid).

  \bigskip

  \textbf{Three} weight matrices (not two like standard ReLU FFN):

  \medskip
  \centering
  \begin{tabular}{lll}
    \toprule
    \textbf{Matrix} & \textbf{Shape} & \textbf{Role} \\
    \midrule
    $\mathbf{W}_{\text{gate}}$ & $(d \times d_{\text{ff}})$ & Gating \\
    $\mathbf{W}_{\text{up}}$   & $(d \times d_{\text{ff}})$ & Value \\
    $\mathbf{W}_{\text{down}}$ & $(d_{\text{ff}} \times d)$ & Projection back \\
    \bottomrule
  \end{tabular}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{SwiGLU --- Worked Example}
  Config: $d = 64$, \; $d_{\text{ff}} = 128$

  \bigskip
  \centering
  \begin{tabular}{llrl}
    \toprule
    \textbf{Matrix} & \textbf{Shape} & \textbf{Params} & \\
    \midrule
    $\mathbf{W}_{\text{gate}}$ & $64 \times 128$ & 8\,192 & \\
    $\mathbf{W}_{\text{up}}$   & $64 \times 128$ & 8\,192 & \\
    $\mathbf{W}_{\text{down}}$ & $128 \times 64$ & 8\,192 & \\
    \midrule
    \textbf{Total per layer}   &                 & \textbf{24\,576} & \\
    \bottomrule
  \end{tabular}

  \bigskip
  Note: $3 \times d \times d_{\text{ff}} = 3 \times 64 \times 128 = 24\,576$
\end{frame}

\section{Design Philosophy}
% ================================================================

% ---------------------------------------------------------------
\begin{frame}{Why So Small?}
  \begin{enumerate}
    \item \textbf{Tiny vocabulary} --- action IDs ($V{=}23$)
          vs.\ text ($V \sim 50\,000$)
    \item \textbf{Weight-tied} output projection
          --- no extra $d \times V$ matrix
    \item \textbf{GQA} with $n_{\text{kv}}{=}1$
          --- KV projections reduced by \alert{75\,\%}
    \item \textbf{SwiGLU} gives better capacity per parameter
          than plain ReLU FFN
    \item \textbf{Designed for the browser}
          --- 286\,KB checkpoint fits in memory
  \end{enumerate}

  \bigskip
  \centering
  \begin{tikzpicture}
    \node[draw, rounded corners=6pt, fill=lightbg,
          inner sep=10pt, font=\large]
      {71\,K params $\;\longleftrightarrow\;$
       286\,KB checkpoint $\;\longleftrightarrow\;$
       runs in browser};
  \end{tikzpicture}
\end{frame}
